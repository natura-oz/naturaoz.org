{{/* Image shortcode with Fancybox support */}} {{- $src := .Get "src" -}} {{- $page := .Page -}} {{/* Find the image resource */}} {{- $img := $page.Resources.GetMatch $src -}} {{- if not $img -}} {{/* Try getting from root */}} {{- $img = resources.Get $src -}} {{- end -}} {{- if not $img -}}

Image not found: {{ $src }}

{{- else -}} {{- $thumbWidth := int (.Get "thumbWidth" | default 800) -}} {{- $thumbHeight := int (.Get "thumbHeight" | default 0) -}} {{- $class := .Get "class" | default "content-image" -}} {{- $caption := .Get "caption" -}} {{- $alt := .Get "alt" | default ($img.Title | default $img.Name) -}} {{- $group := .Get "group" | default "default" -}} {{- /* Generate thumbnail if dimensions specified */}} {{- if or (ne $thumbWidth 0) (ne $thumbHeight 0) -}} {{- $img = $img.Resize (printf "%dx%d" $thumbWidth $thumbHeight) -}} {{- end -}} {{- $captionText := "" -}} {{- if $caption -}} {{- $captionText = $caption -}} {{- else if .Inner -}} {{- $captionText = .Inner -}} {{- else if $img.Params.caption -}} {{- $captionText = $img.Params.caption -}} {{- else if $img.Title -}} {{- $captionText = $img.Title -}} {{- end -}} {{- end -}}
